<body>

    <h2>Taryfikator RP</h2>
    <input type="text" id="search" placeholder="Szukaj (np. pasy, prędkość)..." oninput="filterData()">
    <div id="results"></div>

    <script>
        // Twoja baza danych (działa całkowicie bez internetu)
        const database = [
            { code: "K001", desc: "Przekroczenie prędkości o 10-20 km/h", fine: "50 zł / 2 pkt" },
            { code: "K002", desc: "Brak zapiętych pasów bezpieczeństwa", fine: "100 zł / 5 pkt" },
            { code: "K003", desc: "Rozmowa przez telefon podczas jazdy", fine: "500 zł / 12 pkt" },
            { code: "K004", desc: "Przejechanie na czerwonym świetle", fine: "500 zł / 15 pkt" }
        ];

        function filterData() {
            const query = document.getElementById('search').value.toLowerCase();
            const resultsDiv = document.getElementById('results');
            resultsDiv.innerHTML = '';
            
            const filtered = database.filter(item => 
                item.desc.toLowerCase().includes(query) || item.code.toLowerCase().includes(query)
            );

            if(filtered.length === 0) {
                resultsDiv.innerHTML = '<p style="text-align:center; color:#777;">Brak wyników</p>';
                return;
            }

            filtered.forEach(item => {
                resultsDiv.innerHTML += `
                    <div class="item">
                        <span class="code">${item.code}</span><br>
                        ${item.desc}<br>
                        Mandat: <span class="fine">${item.fine}</span>
                    </div>`;
            });
        }

        // Uruchomienie wyszukiwania przy starcie
        filterData();

        // Rejestracja Service Workera, żeby aplikacja działała offline
        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('sw.js')
                .then(() => console.log("Service Worker działa!"))
                .catch((err) => console.log("Błąd SW:", err));
        }
    </script>
